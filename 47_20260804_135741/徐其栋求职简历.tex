% ======================== 项目经历 ========================
\section{\faProjectDiagram\hspace{0.4em}项目经历}

\projectblock{四足物联网机器狗控制系统}{2025.03 -- 2025.04}{技术栈：STM32F407ZE\hspace{0.3em}|\hspace{0.3em}PCA9685\hspace{0.3em}|\hspace{0.3em}MPU6050\hspace{0.3em}|\hspace{0.3em}ESP32\hspace{0.3em}|\hspace{0.3em}WS2812\hspace{0.3em}|\hspace{0.3em}I2C / UART\hspace{0.3em}|\hspace{0.3em}二维逆运动学}

\begin{itemize}[leftmargin=1.5em, itemsep=1pt, parsep=0pt, topsep=2pt]
    \item \textbf{多舵机运动控制}：用 STM32F407ZE 通过 I2C 控制 PCA9685，扩展出 16 路 12 位 PWM，驱动四条腿共 8 个舵机关节，实现站立、前进、后退、转向等 9 种动作与姿态。
    \item \textbf{逆运动学与平滑控制}：把腿简化为大腿、小腿两根连杆，由足端目标坐标 $(x, y)$ 反算两个关节角度；加入目标位置能否到达的判断和 0\textdegree--180\textdegree{} 角度限制，并把目标角度按 1\textdegree{} 一步拆开下发，避免舵机突然大角度转动造成的抖动。
    \item \textbf{姿态检测与多模交互}：通过 I2C 读取 MPU6050 的三轴加速度和角速度，判断机器狗是否倾倒并反馈状态；USART2 接蓝牙模块、由网页端下发动作指令，ESP32 经 UART4 接入语音控制，WS2812 灯显示当前运行状态。
\end{itemize}

    \item \textbf{四模式控制与 PWM 调光}：实现自动、手动、蓝牙、语音 4 种控制模式；自动模式采集环境光 ADC 数据，通过 TIM2 PWM 完成多档亮度调节，并根据不同输入切换对应控制逻辑。
    \item \textbf{多传感器判断与人机交互}：结合红外在位检测和 HC-SR04 距离数据，实现无人自动关灯与近距离报警；OLED 显示亮度档位及模式，USART2/3 分别接收蓝牙和语音控制指令，形成从采集、判断、调光到显示的完整流程。
\end{itemize}
